% Compatibility shim for ELA's official 2016 SIAM class on recent LaTeX.
% Load this file BEFORE \documentclass:
%   \input{siamart1116-compat}
%   \documentclass[10pt,twoside]{siamart1116}
%
% siamart1116 redefines kernel versions of \label and \refstepcounter.
% Since the 2024-11 LaTeX kernel update, this can conflict with the
% hyperref/cleveref support loaded by the class.  Preserve the kernel
% definitions and restore them immediately before hyperref is loaded.
% The conditional leaves older LaTeX installations unchanged.

\ifdefined\AddToHook
  % The current kernel adds \@currentcounter to \eqnarray.  The 2016 class
  % tests against the preceding kernel definition before applying its own
  % spacing correction; bridge that one-line version difference after
  % ntheorem is loaded.  The manuscript itself does not use eqnarray.
  \makeatletter
  \def\ELALegacyEqnarray{%
     \stepcounter{equation}%
     \def\@currentlabel{\p@equation\theequation}%
     \global\@eqnswtrue
     \m@th
     \global\@eqcnt\z@
     \tabskip\@centering
     \let\\\@eqncr
     $$\everycr{}\halign to\displaywidth\bgroup
         \hskip\@centering$\displaystyle\tabskip\z@skip{##}$\@eqnsel
        &\global\@eqcnt\@ne\hskip \tw@\arraycolsep \hfil${##}$\hfil
        &\global\@eqcnt\tw@ \hskip \tw@\arraycolsep
           $\displaystyle{##}$\hfil\tabskip\@centering
        &\global\@eqcnt\thr@@ \hb@xt@\z@\bgroup\hss##\egroup
           \tabskip\z@skip
        \cr
  }
  \AddToHook{package/ntheorem/after}{\let\eqnarray\ELALegacyEqnarray}
  \makeatother
  \let\ELAOriginalLabel\label
  \let\ELAOriginalRefstepcounter\refstepcounter
  \AddToHook{package/hyperref/before}{%
    \let\label\ELAOriginalLabel
    \let\refstepcounter\ELAOriginalRefstepcounter}
\fi
